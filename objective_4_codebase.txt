"""
Multi-dimensional evaluation metrics for imbalanced classification, probability calibration,
synthetic data statistical fidelity, and attribution stability.
"""

from typing import Dict, Any, Tuple
import numpy as np
from sklearn.metrics import (
    average_precision_score,
    roc_auc_score,
    recall_score,
    precision_score,
    f1_score,
    balanced_accuracy_score,
    brier_score_loss,
)
from scipy.stats import wasserstein_distance, ks_2samp


def compute_expected_calibration_error(
    y_true: np.ndarray,
    y_prob: np.ndarray,
    n_bins: int = 10,
) -> float:
    """
    Computes Expected Calibration Error (ECE).

    ECE = sum_{b=1}^M (|B_b| / N) * |acc(B_b) - conf(B_b)|
    """
    y_true = np.asarray(y_true)
    y_prob = np.asarray(y_prob)

    bin_boundaries = np.linspace(0.0, 1.0, n_bins + 1)
    ece = 0.0
    total_samples = len(y_true)

    for i in range(n_bins):
        bin_lower = bin_boundaries[i]
        bin_upper = bin_boundaries[i + 1]

        # Select samples within current bin boundary
        in_bin = (y_prob >= bin_lower) & (y_prob < bin_upper if i < n_bins - 1 else y_prob <= bin_upper)
        bin_size = np.sum(in_bin)

        if bin_size > 0:
            bin_acc = np.mean(y_true[in_bin])
            bin_conf = np.mean(y_prob[in_bin])
            ece += (bin_size / total_samples) * np.abs(bin_acc - bin_conf)

    return float(ece)


def compute_g_mean(y_true: np.ndarray, y_pred: np.ndarray) -> float:
    """
    Computes Geometric Mean (G-Mean) of recall across binary classes.
    G-Mean = sqrt(Recall_minority * Recall_majority)
    """
    y_true = np.asarray(y_true)
    y_pred = np.asarray(y_pred)

    classes = np.unique(y_true)
    if len(classes) < 2:
        return 0.0

    recalls = []
    for cls in classes:
        rec = recall_score(y_true == cls, y_pred == cls, zero_division=0)
        recalls.append(rec)

    return float(np.sqrt(np.prod(recalls)))


def compute_distributional_fidelity(
    real_minority: np.ndarray,
    synthetic_minority: np.ndarray,
) -> Dict[str, float]:
    """
    Computes statistical divergence between real minority samples and synthetic instances.

    Returns mean Wasserstein distance and fraction of features passing Kolmogorov-Smirnov test (p >= 0.05).
    """
    real = np.asarray(real_minority, dtype=np.float64)
    synth = np.asarray(synthetic_minority, dtype=np.float64)

    n_features = real.shape[1]
    w_distances = []
    ks_p_values = []

    for j in range(n_features):
        wd = wasserstein_distance(real[:, j], synth[:, j])
        w_distances.append(wd)

        ks_res = ks_2samp(real[:, j], synth[:, j])
        ks_p_values.append(ks_res.pvalue)

    mean_wasserstein = float(np.mean(w_distances))
    ks_pass_ratio = float(np.mean(np.array(ks_p_values) >= 0.05))

    return {
        "mean_wasserstein_distance": mean_wasserstein,
        "ks_test_pass_ratio": ks_pass_ratio,
    }


def evaluate_multi_dimensional_metrics(
    model: Any,
    X_test: np.ndarray,
    y_test: np.ndarray,
) -> Dict[str, float]:
    """
    Computes multi-dimensional metric assessment matrix for a fitted classifier model on a test set.

    Metrics include:
      - PR-AUC (Average Precision)
      - ROC-AUC
      - Minority Recall
      - Minority Precision
      - F1 Score
      - Balanced Accuracy
      - G-Mean
      - Expected Calibration Error (ECE)
      - Brier Score
    """
    X_test_arr = np.asarray(X_test, dtype=np.float64)
    y_test_arr = np.asarray(y_test)

    classes, counts = np.unique(y_test_arr, return_counts=True)
    minority_class = classes[np.argmin(counts)]

    # Predictions
    y_pred = model.predict(X_test_arr)

    # Probabilities
    if hasattr(model, "predict_proba"):
        y_prob_raw = model.predict_proba(X_test_arr)
        if y_prob_raw.ndim == 2 and y_prob_raw.shape[1] == 2:
            # Get probability of minority class
            min_idx = list(classes).index(minority_class)
            y_prob = y_prob_raw[:, min_idx]
        else:
            y_prob = y_prob_raw.ravel()
    else:
        y_prob = y_pred.astype(float)

    y_binary = (y_test_arr == minority_class).astype(int)

    # Classification Metrics
    pr_auc = float(average_precision_score(y_binary, y_prob)) if len(np.unique(y_binary)) > 1 else 0.0
    roc_auc = float(roc_auc_score(y_binary, y_prob)) if len(np.unique(y_binary)) > 1 else 0.0
    rec = float(recall_score(y_binary, (y_pred == minority_class).astype(int), zero_division=0))
    prec = float(precision_score(y_binary, (y_pred == minority_class).astype(int), zero_division=0))
    f1 = float(f1_score(y_binary, (y_pred == minority_class).astype(int), zero_division=0))
    bal_acc = float(balanced_accuracy_score(y_binary, (y_pred == minority_class).astype(int)))
    g_mean = compute_g_mean(y_test_arr, y_pred)

    # Probability Calibration
    ece = compute_expected_calibration_error(y_binary, y_prob)
    brier = float(brier_score_loss(y_binary, y_prob))

    return {
        "pr_auc": pr_auc,
        "roc_auc": roc_auc,
        "minority_recall": rec,
        "minority_precision": prec,
        "minority_f1": f1,
        "balanced_accuracy": bal_acc,
        "g_mean": g_mean,
        "ece": ece,
        "brier_score": brier,
    }

"""
Feature Attribution Stability Analysis.

Evaluates whether synthetic oversampling stabilizes or perturbs feature attributions
across cross-validation folds using rank correlation coefficients:
- Kendall's Tau (pairwise & mean)
- Spearman's Rho (pairwise & mean)
- Top-k Rank Overlap (Jaccard similarity of top-k features across folds)
"""

from typing import List, Dict, Any, Tuple
import numpy as np
from scipy.stats import kendalltau, spearmanr


def compute_pairwise_rank_stability(
    rankings: List[np.ndarray],
) -> Dict[str, float]:
    """
    Computes pairwise Kendall's Tau and Spearman's Rho correlation across folds.

    Parameters
    ----------
    rankings : list of np.ndarray
        List of 1D arrays representing feature importance vectors per fold.

    Returns
    -------
    dict with mean and standard deviation of Kendall's tau and Spearman's rho.
    """
    if len(rankings) < 2:
        return {
            "kendall_tau_mean": 0.0,
            "kendall_tau_std": 0.0,
            "spearman_rho_mean": 0.0,
            "spearman_rho_std": 0.0,
        }

    kendall_taus = []
    spearman_rhos = []

    for i in range(len(rankings)):
        for j in range(i + 1, len(rankings)):
            r_i, r_j = rankings[i], rankings[j]
            tau, _ = kendalltau(r_i, r_j)
            rho, _ = spearmanr(r_i, r_j)
            if not np.isnan(tau):
                kendall_taus.append(tau)
            if not np.isnan(rho):
                spearman_rhos.append(rho)

    tau_mean = float(np.mean(kendall_taus)) if kendall_taus else 0.0
    tau_std = float(np.std(kendall_taus)) if kendall_taus else 0.0
    rho_mean = float(np.mean(spearman_rhos)) if spearman_rhos else 0.0
    rho_std = float(np.std(spearman_rhos)) if spearman_rhos else 0.0

    return {
        "kendall_tau_mean": tau_mean,
        "kendall_tau_std": tau_std,
        "spearman_rho_mean": rho_mean,
        "spearman_rho_std": rho_std,
    }


def compute_top_k_rank_overlap(
    rankings: List[np.ndarray],
    k: int = 5,
) -> float:
    """
    Computes mean pairwise Jaccard index of top-k ranked features across folds.

    Parameters
    ----------
    rankings : list of np.ndarray
        Feature attribution vectors per fold.
    k : int, default=5
        Number of top features to compare.

    Returns
    -------
    mean_jaccard : float between 0.0 and 1.0.
    """
    if len(rankings) < 2:
        return 1.0

    top_k_sets = []
    for r in rankings:
        top_indices = set(np.argsort(r)[::-1][:k])
        top_k_sets.append(top_indices)

    jaccards = []
    for i in range(len(top_k_sets)):
        for j in range(i + 1, len(top_k_sets)):
            s1, s2 = top_k_sets[i], top_k_sets[j]
            inter = len(s1.intersection(s2))
            union = len(s1.union(s2))
            jaccard = inter / union if union > 0 else 0.0
            jaccards.append(jaccard)

    return float(np.mean(jaccards)) if jaccards else 0.0


def evaluate_explanation_stability(
    rankings: List[np.ndarray],
    top_k: int = 5,
) -> Dict[str, float]:
    """
    Comprehensive explanation stability evaluation summary.
    """
    res = compute_pairwise_rank_stability(rankings)
    res["top_k_overlap"] = compute_top_k_rank_overlap(rankings, k=top_k)
    return res

"""
Synthetic Data Statistical Fidelity Evaluation.

Quantifies distributional divergence between real minority samples and synthetic instances:
- Marginal Wasserstein Distance (mean & maximum)
- Marginal Kolmogorov-Smirnov (KS) two-sample tests
- Covariance & Correlation Matrix Preservation
"""

from typing import Dict, Any, List
import numpy as np
from scipy.stats import wasserstein_distance, ks_2samp, pearsonr


def compute_synthetic_fidelity(
    real_minority: np.ndarray,
    synthetic_minority: np.ndarray,
) -> Dict[str, float]:
    """
    Evaluates statistical fidelity between real minority points and synthetic instances.

    Parameters
    ----------
    real_minority : np.ndarray of shape (N_real, p)
        Real minority samples from training fold.
    synthetic_minority : np.ndarray of shape (N_synth, p)
        Generated synthetic minority samples.

    Returns
    -------
    dict containing:
      - mean_wasserstein_distance
      - max_wasserstein_distance
      - ks_test_pass_ratio (fraction of features with p >= 0.05)
      - covariance_frobenius_dist
      - correlation_matrix_similarity (Pearson correlation between upper triangles)
    """
    real = np.asarray(real_minority, dtype=np.float64)
    synth = np.asarray(synthetic_minority, dtype=np.float64)

    if real.shape[0] < 2 or synth.shape[0] < 2:
        return {
            "mean_wasserstein_distance": 0.0,
            "max_wasserstein_distance": 0.0,
            "ks_test_pass_ratio": 1.0,
            "covariance_frobenius_dist": 0.0,
            "correlation_matrix_similarity": 1.0,
        }

    n_features = real.shape[1]
    w_distances = []
    ks_pvals = []

    for j in range(n_features):
        wd = wasserstein_distance(real[:, j], synth[:, j])
        w_distances.append(wd)

        ks_stat = ks_2samp(real[:, j], synth[:, j])
        ks_pvals.append(ks_stat.pvalue)

    mean_wd = float(np.mean(w_distances))
    max_wd = float(np.max(w_distances))
    ks_pass = float(np.mean(np.array(ks_pvals) >= 0.05))

    # Covariance preservation
    cov_real = np.cov(real, rowvar=False)
    cov_synth = np.cov(synth, rowvar=False)
    cov_frob = float(np.linalg.norm(cov_real - cov_synth, ord="fro"))

    # Correlation matrix preservation
    corr_real = np.corrcoef(real, rowvar=False)
    corr_synth = np.corrcoef(synth, rowvar=False)

    # Handle NaNs if zero variance
    corr_real = np.nan_to_num(corr_real, nan=0.0)
    corr_synth = np.nan_to_num(corr_synth, nan=0.0)

    triu_indices = np.triu_indices(n_features, k=1)
    if len(triu_indices[0]) > 0:
        tri_real = corr_real[triu_indices]
        tri_synth = corr_synth[triu_indices]
        # Pearson correlation between correlation matrix entries
        if np.std(tri_real) > 1e-8 and np.std(tri_synth) > 1e-8:
            corr_sim, _ = pearsonr(tri_real, tri_synth)
            corr_sim = float(corr_sim)
        else:
            corr_sim = 1.0 - min(1.0, float(np.mean(np.abs(tri_real - tri_synth))))
    else:
        corr_sim = 1.0

    return {
        "mean_wasserstein_distance": mean_wd,
        "max_wasserstein_distance": max_wd,
        "ks_test_pass_ratio": ks_pass,
        "covariance_frobenius_dist": cov_frob,
        "correlation_matrix_similarity": corr_sim,
    }

"""
Visualization and Publication-Quality LaTeX Export Engine.

Generates:
- Benchmark multi-metric comparison bar charts
- Precision-Recall curves & ROC curves
- Calibration reliability diagrams
- Attribution stability heatmaps / comparisons
- Publication-ready LaTeX tables (booktabs formatted)
"""

from pathlib import Path
from typing import Dict, Any, List, Optional
import numpy as np
import pandas as pd
import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt


def plot_benchmark_metric_comparison(
    df_results: pd.DataFrame,
    metric_col: str = "pr_auc_mean",
    metric_label: str = "PR-AUC (Average Precision)",
    output_path: Optional[str] = None,
    title: Optional[str] = None,
) -> plt.Figure:
    """
    Plots grouped bar chart comparing oversamplers across classifiers or datasets.
    """
    fig, ax = plt.subplots(figsize=(10, 6))

    classifiers = df_results["classifier"].unique()
    oversamplers = df_results["oversampler"].unique()

    x = np.arange(len(oversamplers))
    width = 0.8 / len(classifiers)

    for i, clf in enumerate(classifiers):
        sub_df = df_results[df_results["classifier"] == clf]
        # Align with oversamplers order
        vals = []
        errors = []
        for o in oversamplers:
            row = sub_df[sub_df["oversampler"] == o]
            if len(row) > 0:
                vals.append(row[metric_col].values[0])
                std_col = metric_col.replace("_mean", "_std")
                errors.append(row[std_col].values[0] if std_col in row else 0.0)
            else:
                vals.append(0.0)
                errors.append(0.0)

        offset = (i - len(classifiers) / 2 + 0.5) * width
        ax.bar(
            x + offset,
            vals,
            width,
            yerr=errors,
            capsize=3,
            label=clf,
            alpha=0.9,
            edgecolor="black",
            linewidth=0.7,
        )

    ax.set_ylabel(metric_label, fontsize=12, fontweight="bold")
    ax.set_title(title or f"Empirical Benchmark Comparison: {metric_label}", fontsize=14, fontweight="bold")
    ax.set_xticks(x)
    ax.set_xticklabels(oversamplers, rotation=25, ha="right", fontsize=10)
    ax.legend(title="Classifier", fontsize=10)
    ax.grid(axis="y", linestyle="--", alpha=0.6)
    plt.tight_layout()

    if output_path:
        Path(output_path).parent.mkdir(parents=True, exist_ok=True)
        fig.savefig(output_path, dpi=300)

    return fig


def plot_multi_dimensional_radar(
    df_results: pd.DataFrame,
    oversamplers_to_compare: List[str],
    metrics: List[str] = None,
    metric_labels: List[str] = None,
    output_path: Optional[str] = None,
) -> plt.Figure:
    """
    Generates radar / polar chart comparing multi-dimensional trade-offs
    (Utility, Calibration, Stability).
    """
    if metrics is None:
        metrics = ["pr_auc_mean", "minority_f1_mean", "balanced_accuracy_mean", "attribution_kendall_tau_mean"]
    if metric_labels is None:
        metric_labels = ["PR-AUC", "Minority F1", "Balanced Acc", "Attribution Tau"]

    N = len(metrics)
    angles = np.linspace(0, 2 * np.pi, N, endpoint=False).tolist()
    angles += angles[:1]  # Close loop

    fig, ax = plt.subplots(figsize=(8, 8), subplot_kw=dict(polar=True))

    for o_name in oversamplers_to_compare:
        sub = df_results[df_results["oversampler"] == o_name]
        if len(sub) == 0:
            continue
        vals = [sub[m].mean() for m in metrics]
        # Invert ECE or Brier if present so outward is always better
        vals_plot = []
        for m, v in zip(metrics, vals):
            if "ece" in m or "brier" in m:
                vals_plot.append(max(0.0, 1.0 - v))
            else:
                vals_plot.append(v)
        vals_plot += vals_plot[:1]

        ax.plot(angles, vals_plot, linewidth=2, label=o_name)
        ax.fill(angles, vals_plot, alpha=0.15)

    ax.set_xticks(angles[:-1])
    ax.set_xticklabels(metric_labels, fontsize=11, fontweight="bold")
    ax.set_title("Multi-Dimensional Performance Trade-Offs", size=14, y=1.08, fontweight="bold")
    ax.legend(loc="upper right", bbox_to_anchor=(1.3, 1.1), fontsize=10)
    plt.tight_layout()

    if output_path:
        Path(output_path).parent.mkdir(parents=True, exist_ok=True)
        fig.savefig(output_path, dpi=300)

    return fig


def generate_latex_table(
    df_results: pd.DataFrame,
    caption: str = "Empirical Benchmark Results across Imbalanced Oversampling Techniques.",
    label: str = "tab:benchmark_results",
    selected_cols: Optional[List[str]] = None,
) -> str:
    """
    Converts benchmark DataFrame into publication-ready LaTeX table using booktabs.
    """
    if selected_cols is None:
        selected_cols = [
            "oversampler",
            "classifier",
            "pr_auc_mean",
            "roc_auc_mean",
            "minority_f1_mean",
            "ece_mean",
            "attribution_kendall_tau_mean",
        ]

    df_sub = df_results[[c for c in selected_cols if c in df_results.columns]].copy()

    rename_map = {
        "oversampler": "Oversampler",
        "classifier": "Classifier",
        "pr_auc_mean": "PR-AUC",
        "roc_auc_mean": "ROC-AUC",
        "minority_f1_mean": "F1-Score",
        "ece_mean": "ECE $\\downarrow$",
        "attribution_kendall_tau_mean": "Kendall $\\tau$",
    }
    df_sub = df_sub.rename(columns=rename_map)

    # Format numeric floats to 4 decimal places
    for col in df_sub.columns:
        if df_sub[col].dtype in (np.float64, np.float32, float):
            df_sub[col] = df_sub[col].map(lambda x: f"{x:.4f}")

    col_fmt = "l" * len(df_sub.columns)
    headers = " & ".join(df_sub.columns) + " \\\\"
    
    rows = []
    for _, row in df_sub.iterrows():
        row_str = " & ".join(str(val) for val in row.values) + " \\\\"
        rows.append(row_str)
    
    body = "\n".join(rows)

    latex_code = (
        "\\begin{table}[htbp]\n"
        "\\centering\n"
        f"\\caption{{{caption}}}\n"
        f"\\label{{{label}}}\n"
        "\\begin{tabular}{" + col_fmt + "}\n"
        "\\toprule\n"
        f"{headers}\n"
        "\\midrule\n"
        f"{body}\n"
        "\\bottomrule\n"
        "\\end{tabular}\n"
        "\\end{table}\n"
    )
    return latex_code

"""
Leak-Free Cross-Validation Pipeline ensuring zero data leakage during feature scaling,
SHAP attribution, synthetic sample generation, and model training.
"""

from typing import Dict, Any, List, Optional, Tuple
import numpy as np
from sklearn.base import BaseEstimator, ClassifierMixin

class LeakFreePipeline(BaseEstimator, ClassifierMixin):
    """
    Leak-Free Pipeline encapsulating:
      1. Training set feature scaling (StandardScaler fit on train only)
      2. In-fold synthetic oversampling (SHAP-Guided or baseline)
      3. Final downstream classifier fit
      4. Prediction on untouched test set

    Parameters
    ----------
    oversampler : BaseEstimator
        Instance of oversampler (e.g., SHAPFilteredSMOTE, SHAPWeightedSMOTE, or baseline).
    classifier : BaseEstimator
        Final downstream classification model (e.g. LogisticRegression, RandomForest, XGBoost).
    scale_features : bool, default=True
        Whether to fit StandardScaler on training fold prior to oversampling.
    """

    def __init__(
        self,
        oversampler: BaseEstimator,
        classifier: BaseEstimator,
        scale_features: bool = True,
    ):
        self.oversampler = oversampler
        self.classifier = classifier
        self.scale_features = scale_features

    def fit(self, X: np.ndarray, y: np.ndarray) -> "LeakFreePipeline":
        """
        Fit pipeline strictly on inner training fold data.
        """
        X_arr = np.asarray(X, dtype=np.float64)
        y_arr = np.asarray(y)

        # 1. Fit scaler on training fold only
        if self.scale_features:
            self.scaler_ = StandardScaler()
            X_scaled = self.scaler_.fit_transform(X_arr)
        else:
            self.scaler_ = None
            X_scaled = X_arr.copy()

        # 2. Oversample training fold
        if hasattr(self.oversampler, "fit_resample"):
            X_resampled, y_resampled = self.oversampler.fit_resample(X_scaled, y_arr)
        else:
            X_resampled, y_resampled = X_scaled, y_arr

        # 3. Fit final classifier on resampled dataset
        self.classifier_ = self.classifier
        self.classifier_.fit(X_resampled, y_resampled)

        return self

    def predict(self, X: np.ndarray) -> np.ndarray:
        X_arr = np.asarray(X, dtype=np.float64)
        if self.scaler_ is not None:
            X_arr = self.scaler_.transform(X_arr)
        return self.classifier_.predict(X_arr)

    def predict_proba(self, X: np.ndarray) -> np.ndarray:
        X_arr = np.asarray(X, dtype=np.float64)
        if self.scaler_ is not None:
            X_arr = self.scaler_.transform(X_arr)
        return self.classifier_.predict_proba(X_arr)


def run_leak_free_benchmark(
    X: np.ndarray,
    y: np.ndarray,
    oversampler: BaseEstimator,
    classifier: BaseEstimator,
    n_splits: int = 5,
    n_repeats: int = 1,
    random_state: int = 42,
    scale_features: bool = True,
) -> Dict[str, Any]:
    """
    Executes leak-free Stratified Repeated Nested Cross-Validation.

    Evaluates downstream model performance, probability calibration,
    and calculates SHAP feature attribution stability (Kendall's Tau & Spearman Rho)
    across test folds.

    Parameters
    ----------
    X : np.ndarray of shape (N, p)
        Dataset feature matrix.
    y : np.ndarray of shape (N,)
        Target label vector.
    oversampler : BaseEstimator
        Oversampler instance.
    classifier : BaseEstimator
        Downstream classifier instance.
    n_splits : int, default=5
        Number of cross-validation splits.
    n_repeats : int, default=1
        Number of cross-validation repeats.
    random_state : int, default=42
        Random seed.
    scale_features : bool, default=True
        Whether to scale features inside fold.

    Returns
    -------
    results : dict
        Aggregated evaluation metrics, fold-level scores, and feature attribution stability.
    """
    X_arr = np.asarray(X, dtype=np.float64)
    y_arr = np.asarray(y)

    rskf = RepeatedStratifiedKFold(
        n_splits=n_splits,
        n_repeats=n_repeats,
        random_state=random_state,
    )

    fold_metrics: List[Dict[str, float]] = []
    shap_rankings: List[np.ndarray] = []

    for fold_idx, (train_idx, test_idx) in enumerate(rskf.split(X_arr, y_arr)):
        X_train, X_test = X_arr[train_idx], X_arr[test_idx]
        y_train, y_test = y_arr[train_idx], y_arr[test_idx]

        pipeline = LeakFreePipeline(
            oversampler=oversampler,
            classifier=classifier,
            scale_features=scale_features,
        )
        pipeline.fit(X_train, y_train)

        # Evaluate metrics on untouched test set
        metrics = evaluate_multi_dimensional_metrics(pipeline, X_test, y_test)
        fold_metrics.append(metrics)

        # Compute SHAP feature rankings on test fold predictions for stability analysis
        try:
            X_test_scaled = pipeline.scaler_.transform(X_test) if pipeline.scaler_ else X_test
            clf = pipeline.classifier_

            if hasattr(clf, "feature_importances_") or "Tree" in type(clf).__name__ or "Forest" in type(clf).__name__ or "XGB" in type(clf).__name__:
                explainer = shap.TreeExplainer(clf)
                sv = explainer.shap_values(X_test_scaled)
            elif hasattr(clf, "coef_"):
                X_train_scaled = pipeline.scaler_.transform(X_train) if pipeline.scaler_ else X_train
                explainer = shap.LinearExplainer(clf, X_train_scaled)
                sv = explainer.shap_values(X_test_scaled)
            else:
                explainer = shap.Explainer(clf.predict_proba, X_test_scaled)
                sv = explainer(X_test_scaled).values

            if isinstance(sv, list):
                sv = sv[1] if len(sv) > 1 else sv[0]
            elif isinstance(sv, np.ndarray) and sv.ndim == 3:
                sv = sv[:, :, 1]

            masv_fold = np.mean(np.abs(sv), axis=0)
            shap_rankings.append(masv_fold)
        except Exception:
            pass

    # Aggregate fold metrics
    metric_keys = fold_metrics[0].keys()
    summary = {}
    for k in metric_keys:
        vals = [m[k] for m in fold_metrics]
        summary[f"{k}_mean"] = float(np.mean(vals))
        summary[f"{k}_std"] = float(np.std(vals))

    # Calculate attribution stability across folds (pairwise Kendall's Tau and Spearman Rho)
    kendall_taus = []
    spearman_rhos = []
    if len(shap_rankings) >= 2:
        for i in range(len(shap_rankings)):
            for j in range(i + 1, len(shap_rankings)):
                tau, _ = kendalltau(shap_rankings[i], shap_rankings[j])
                rho, _ = spearmanr(shap_rankings[i], shap_rankings[j])
                if not np.isnan(tau):
                    kendall_taus.append(tau)
                if not np.isnan(rho):
                    spearman_rhos.append(rho)

    summary["attribution_kendall_tau_mean"] = float(np.mean(kendall_taus)) if kendall_taus else 0.0
    summary["attribution_spearman_rho_mean"] = float(np.mean(spearman_rhos)) if spearman_rhos else 0.0
    summary["n_folds_evaluated"] = len(fold_metrics)

    return summary

def instantiate_oversampler(name: str, random_state: int = 42):
    """Maps name to oversampler instance."""
    name_clean = name.lower().strip()
    if "shap-filtered" in name_clean or "shap_filtered" in name_clean or "filtered" in name_clean:
        return SHAPFilteredSMOTE(cumulative_threshold=0.85, random_state=random_state)
    elif "shap-weighted" in name_clean or "shap_weighted" in name_clean or "masv" in name_clean:
        return SHAPWeightedSMOTE(random_state=random_state)
    elif "raw" in name_clean or "none" in name_clean:
        return get_baseline_oversampler("raw")
    elif "ros" in name_clean or "random oversampling" in name_clean:
        return get_baseline_oversampler("ros", random_state=random_state)
    elif "borderline" in name_clean:
        return get_baseline_oversampler("borderline_smote", random_state=random_state)
    elif "adasyn" in name_clean:
        return get_baseline_oversampler("adasyn", random_state=random_state)
    elif "smote-enn" in name_clean or "smote_enn" in name_clean:
        return get_baseline_oversampler("smote_enn", random_state=random_state)
    elif "smote-tomek" in name_clean or "smote_tomek" in name_clean:
        return get_baseline_oversampler("smote_tomek", random_state=random_state)
    elif "smote" in name_clean:
        return get_baseline_oversampler("smote", random_state=random_state)
    else:
        return get_baseline_oversampler(name, random_state=random_state)


def measure_sampler_fidelity(sampler_instance, X: np.ndarray, y: np.ndarray) -> Dict[str, float]:
    """Measures statistical fidelity between real minority points and synthesized points."""
    try:
        classes, counts = np.unique(y, return_counts=True)
        min_class = classes[np.argmin(counts)]
        X_min_real = X[y == min_class]

        X_res, y_res = sampler_instance.fit_resample(X, y)
        n_synth = len(X_res) - len(X)
        if n_synth > 0:
            X_synth = X_res[len(X):]
            return compute_synthetic_fidelity(X_min_real, X_synth)
    except Exception:
        pass
    return {
        "mean_wasserstein_distance": 0.0,
        "max_wasserstein_distance": 0.0,
        "ks_test_pass_ratio": 1.0,
        "covariance_frobenius_dist": 0.0,
        "correlation_matrix_similarity": 1.0,
    }

def generate_feature_importance_weights(X, y, method="shap", random_state=42):
    """Computes normalized feature weights using SHAP, Gini, or Mutual Information."""
    p = X.shape[1]
    if method == "shap":
        weights, masv, _ = compute_masv_weights(X, y, random_state=random_state)
        return weights
    elif method == "gini":
        rf = RandomForestClassifier(n_estimators=50, max_depth=5, random_state=random_state, n_jobs=-1)
        rf.fit(X, y)
        importances = rf.feature_importances_
        sum_imp = np.sum(importances)
        return importances / sum_imp if sum_imp > 0 else np.full(p, 1.0 / p)
    elif method == "mutual_info":
        mi = mutual_info_classif(X, y, random_state=random_state)
        sum_mi = np.sum(mi)
        return mi / sum_mi if sum_mi > 0 else np.full(p, 1.0 / p)
    elif method == "unweighted":
        return np.full(p, 1.0 / p)
    else:
        raise ValueError(f"Unknown weighting method: {method}")


class FeatureWeightedSMOTE:
    """Generalized Feature-Weighted SMOTE supporting SHAP, Gini, Mutual Info, or Unweighted."""
    def __init__(self, method="shap", k_neighbors=5, random_state=42):
        self.method = method
        self.k_neighbors = k_neighbors
        self.random_state = random_state

    def fit_resample(self, X, y):
        X_arr = np.asarray(X, dtype=np.float64)
        y_arr = np.asarray(y)
        rng = np.random.default_rng(self.random_state)

        weights = generate_feature_importance_weights(X_arr, y_arr, method=self.method, random_state=self.random_state)
        self.feature_weights_ = weights

        classes, counts = np.unique(y_arr, return_counts=True)
        minority_class = classes[np.argmin(counts)]
        majority_class = classes[np.argmax(counts)]

        minority_mask = y_arr == minority_class
        X_min = X_arr[minority_mask]
        n_min = len(X_min)
        n_to_gen = max(0, np.sum(y_arr == majority_class) - n_min)

        if n_to_gen == 0 or n_min < 2:
            return X_arr.copy(), y_arr.copy()

        # Weighted distance search via sqrt(w) projection
        sqrt_w = np.sqrt(weights)
        X_min_weighted = X_min * sqrt_w

        k = min(self.k_neighbors, n_min - 1)
        nn = NearestNeighbors(n_neighbors=k + 1, metric="euclidean", n_jobs=-1)
        nn.fit(X_min_weighted)
        neighbors_idx = nn.kneighbors(X_min_weighted, return_distance=False)[:, 1:]

        synthetic = np.zeros((n_to_gen, X_arr.shape[1]), dtype=np.float64)
        for i in range(n_to_gen):
            base_idx = rng.integers(0, n_min)
            nn_choice = rng.integers(0, k)
            neighbor_idx = neighbors_idx[base_idx, nn_choice]
            lambd = rng.uniform(0.0, 1.0)
            synthetic[i] = X_min[base_idx] + lambd * (X_min[neighbor_idx] - X_min[base_idx])

        X_res = np.vstack([X_arr, synthetic])
        y_res = np.concatenate([y_arr, np.full(n_to_gen, minority_class, dtype=y_arr.dtype)])
        return X_res, y_res

def run_geometric_proof(output_dir: Path):

    # Impose 90:10 imbalance ratio
    maj_idx = np.where(y_raw == 0)[0]
    min_idx = np.where(y_raw == 1)[0][:35]  # 35 minority points vs 250 majority
    keep_idx = np.concatenate([maj_idx, min_idx])
    X = X_raw[keep_idx]
    y = y_raw[keep_idx]

    # Add 2 pure noise dimensions to evaluate feature agnosticism
    rng = np.random.default_rng(42)
    noise = rng.normal(0, 1, size=(len(X), 2))
    X_high = np.hstack([X, noise])

    samplers = {
        "Raw (No Balancing)": RawNoBalancing(),
        "Standard SMOTE": ClassicalSMOTE(k_neighbors=4, random_state=42),
        "Borderline-SMOTE": BorderlineSMOTEWrapper(k_neighbors=4, m_neighbors=6, random_state=42),
        "SHAP-Weighted MASV-SMOTE": SHAPWeightedSMOTE(k_neighbors=4, random_state=42),
    }

    fig, axes = plt.subplots(1, 4, figsize=(20, 5))
    x_min, x_max = X[:, 0].min() - 0.5, X[:, 0].max() + 0.5
    y_min, y_max = X[:, 1].min() - 0.5, X[:, 1].max() + 0.5
    xx, yy = np.meshgrid(np.linspace(x_min, x_max, 100), np.linspace(y_min, y_max, 100))
    grid_2d = np.c_[xx.ravel(), yy.ravel()]
    # Zero pad noise for grid evaluation
    grid_high = np.hstack([grid_2d, np.zeros((len(grid_2d), 2))])

    for ax, (title, sampler) in zip(axes, samplers.items()):
        X_res, y_res = sampler.fit_resample(X_high, y)
        n_synth = len(X_res) - len(X_high)

        clf = RandomForestClassifier(n_estimators=50, max_depth=5, random_state=42)
        clf.fit(X_res, y_res)
        Z = clf.predict_proba(grid_high)[:, 1].reshape(xx.shape)

        # Plot decision contour
        ax.contourf(xx, yy, Z, levels=np.linspace(0, 1, 11), cmap="RdBu_r", alpha=0.6)
        ax.contour(xx, yy, Z, levels=[0.5], colors="black", linewidths=1.5, linestyles="--")

        # Plot original points
        ax.scatter(X[y == 0, 0], X[y == 0, 1], c="#2b5c8f", marker="o", s=25, label="Majority Class (Real)", alpha=0.7)
        ax.scatter(X[y == 1, 0], X[y == 1, 1], c="#d9381e", marker="^", s=50, edgecolors="black", label="Minority Class (Real)")

        # Plot synthetic points
        if n_synth > 0:
            X_synth = X_res[len(X_high):]
            ax.scatter(X_synth[:, 0], X_synth[:, 1], c="#ffc107", marker="*", s=60, edgecolors="black", label=f"Synthetic (n={n_synth})")

        ax.set_title(title, fontsize=13, fontweight="bold")
        ax.set_xlim(x_min, x_max)
        ax.set_ylim(y_min, y_max)
        ax.grid(True, linestyle=":", alpha=0.5)

    axes[0].legend(loc="lower left", fontsize=9, framealpha=0.9)
    plt.suptitle("Proof of Objective 1 & 2: Synthetic Placement and Decision Manifold Comparison", fontsize=15, fontweight="bold", y=1.02)
    plt.tight_layout()

def run_data_leakage_proof(output_dir: Path):

    t_stat, p_val_ttest = stats.ttest_rel(shap_weighted_pr, baseline_pr)

    # Attribution stability (Kendall's tau pairwise across folds)
    taus_per_sampler = {}
    for s_name, rankings in tau_stability.items():
        t_vals = []
        for i in range(len(rankings)):
            for j in range(i + 1, len(rankings)):
                tau, _ = stats.kendalltau(rankings[i], rankings[j])
                if not np.isnan(tau):
                    t_vals.append(tau)
        taus_per_sampler[s_name] = t_vals

    fig, (ax_box, ax_bar) = plt.subplots(1, 2, figsize=(14, 5))

    # Boxplot of PR-AUC across folds
    box_data = [[m["pr_auc"] for m in fold_results[s]] for s in samplers]
    bp = ax_box.boxplot(box_data, patch_artist=True, labels=[s.replace(" ", "\n") for s in samplers])
    palette = ["#7f8c8d", "#e74c3c", "#9b59b6", "#2980b9", "#27ae60"]
    for patch, color in zip(bp["boxes"], palette):
        patch.set_facecolor(color)
        patch.set_alpha(0.8)

    ax_box.set_ylabel("PR-AUC (5-Fold CV)", fontsize=11, fontweight="bold")
    ax_box.set_title("Cross-Fold PR-AUC Dispersion\n(Paired Wilcoxon p = {:.4f})".format(p_val_wilcoxon), fontsize=12, fontweight="bold")
    ax_box.grid(axis="y", linestyle=":", alpha=0.6)

    # Feature rank stability bar chart
    tau_means = [np.mean(taus_per_sampler[s]) for s in samplers]
    tau_stds = [np.std(taus_per_sampler[s]) for s in samplers]
    bars = ax_bar.bar([s.replace(" ", "\n") for s in samplers], tau_means, yerr=tau_stds, capsize=4, color=palette, alpha=0.85, edgecolor="black")
    ax_bar.set_ylabel("Pairwise Kendall's Tau Stability", fontsize=11, fontweight="bold")
    ax_bar.set_title("Downstream Explanation Stability Across CV Folds", fontsize=12, fontweight="bold")
    ax_bar.grid(axis="y", linestyle=":", alpha=0.6)
    for bar in bars:
        yval = bar.get_height()
        ax_bar.text(bar.get_x() + bar.get_width() / 2, yval + 0.02, f"{yval:.3f}", ha="center", fontweight="bold")

    plt.suptitle("Proof of Objective 4: Statistical Significance and Feature Attribution Stability", fontsize=14, fontweight="bold", y=1.02)
    plt.tight_layout()

    out_file2 = output_dir / "fig5_statistical_significance_and_stability.png"
    fig.savefig(out_file2, dpi=300, bbox_inches="tight")
    plt.close(fig)
    print(f"    [->] Saved: {out_file2}")

    return {
        "p_val_paired_ttest": float(p_val_ttest),
        "p_val_wilcoxon": float(p_val_wilcoxon),
        "fold_results": fold_results,
        "tau_stability": {s: float(np.mean(taus_per_sampler[s])) for s in samplers},
    }
